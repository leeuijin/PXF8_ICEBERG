# 테스트 환경 구성

CREATE EXTENSION pxf;
CREATE EXTENSION pxf_fdw;

#ICEBERG 카탈로그 데이터베이스 생성

$createdb iceberg_catalog

# PXF 설정
mkdir -p /usr/local/pxf-gp7/clusters/default/servers/iceberg-s3

[gpadmin@mdw iceberg-s3]$ cat iceberg-site.xml

    <property>
        <name>iceberg.catalog</name>

    <property>

        <value>jdbc</value>
    </property>
    <property>
        <name>iceberg.catalog.default.uri</name>
        <value>jdbc:postgresql://[META_DB_IP]]:[META_DB_PORT]/iceberg_catalog</value>

# PXF 시작

$pxf cluster sync
$pxf cluster start
$pxf cluster status

# 외부 서버 생성 및 권한 부여
CREATE SERVER pxf_iceberg_s3 FOREIGN DATA WRAPPER iceberg_pxf_fdw OPTIONS (
    config 'iceberg-s3'
);
ALTER SERVER pxf_iceberg_s3 OWNER TO gpadmin;
CREATE USER MAPPING FOR gpadmin SERVER pxf_iceberg_s3;

# iceberg_s3 외부 테이블 생성
CREATE FOREIGN TABLE public.ext_clickstream_events (
    event_id integer,
    event_type text,
    event_time timestamp without time zone,
    user_id bigint,
    payload text
)
SERVER pxf_iceberg_s3
OPTIONS (
    resource 'analytics.clickstream_events'
);
